// File: /api/log/index.js export async function onRequestPost(context) { const { request, env } = context; // Your Amplitude API Key const AMPLITUDE_API_KEY = '3c8ae1f40635939e730f479418940796'; const AMPLITUDE_ENDPOINT = 'https://api2.amplitude.com/2/httpapi'; try { // 1. Get the data sent from the bookmarklet const requestData = await request.json(); // 2. Prepare the event for Amplitude // We'll create a generic event for any intercepted data const amplitudeEvent = { user_id: requestData.interceptedUserId || 'anonymous_user', // Use a user ID if available event_type: 'Intercepted Bookmarklet Data', event_properties: { interceptedUrl: requestData.interceptedUrl, interceptedMethod: requestData.interceptedMethod, interceptedHeaders: JSON.stringify(requestData.interceptedHeaders), interceptedBody: requestData.interceptedBody, // The actual data payload userAgent: request.headers.get('User-Agent'), timestamp: new Date().toISOString() } }; // 3. Send the event to Amplitude const amplitudeResponse = await fetch(AMPLITUDE_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': '*/*' }, body: JSON.stringify({ api_key: AMPLITUDE_API_KEY, events: [amplitudeEvent] }) }); if (!amplitudeResponse.ok) { const errorText = await amplitudeResponse.text(); console.error('Failed to send to Amplitude:', errorText); // Return an error, but don't expose internal details to the client return new Response(JSON.stringify({ success: false, error: 'Failed to log to analytics' }), { status: 500, headers: { 'Content-Type': 'application/json' }, }); } // 4. Send a success response back to the bookmarklet return new Response(JSON.stringify({ success: true, message: 'Data logged successfully' }), { status: 200, headers: { 'Content-Type': 'application/json' }, }); } catch (err) { console.error('Error in Cloudflare Function:', err); return new Response(JSON.stringify({ success: false, error: 'Internal server error' }), { status: 500, headers: { 'Content-Type': 'application/json' }, }); } }
